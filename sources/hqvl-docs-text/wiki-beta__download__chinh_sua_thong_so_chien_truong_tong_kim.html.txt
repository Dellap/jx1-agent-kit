Bến dưới là đường dẫn đến tập tin qui định các thông số cho chiến trường Tống Kim trong server (máy chủ CentOS).
/home/jxser/server1/script/battles/battlehead.lua

Chi tiết ý nghĩa các thông số
FRAME2TIME = 18;      --18 tương đương 1 giây
BAOMING_TIME = 10     --10 phút thời gian đăng ký
FIGHTING_TIME = 60    -- Thời gian diễn ra trận đấu, mặc định là 60 phút
ANNOUNCE_TIME = 20    -- 20 giây công bố tại chiến trận

BOSS_TIME_MAIN=30    --30 phút sau khi báo danh xuất hiện nguyên soái
VANISHGAME_TIME_MAIN=20    --20 phút cuối xuất hiện nguyên soái tiếp theo

TIMER_1 = ANNOUNCE_TIME * FRAME2TIME;    --20 giây công bố tại chiến trận
TIMER_2 = (FIGHTING_TIME + BAOMING_TIME) * 60  * FRAME2TIME;  --Thời gian chiến đấu 1H
RUNGAME_TIME = BAOMING_TIME * 60 * FRAME2TIME / TIMER_1;    --10 phút sau khi đăng ký, tự động bước vào giai đoạn chiến đấu
GO_TIME =  BAOMING_TIME * 60 * FRAME2TIME  / TIMER_1;       --Thời gian áp dụng là 10 phút sẽ RunMission

SONGJIN_SIGNUP_FEES = 10000                 --Phí đăng ký
